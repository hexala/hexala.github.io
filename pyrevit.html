def last_row(schedule):
	key_name = 0
	rows = UnwrapElement(FilteredElementCollector(doc, schedule.Id).ToElements())
	for j in rows:
		if int(j.get_Parameter(BuiltInParameter.REF_TABLE_ELEM_NAME).AsString()) > key_name:
			key_name = int(j.get_Parameter(BuiltInParameter.REF_TABLE_ELEM_NAME).AsString())
			row = doc.GetElement(j.Id)
	return row
